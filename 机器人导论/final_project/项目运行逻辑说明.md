```python
    def __init__(self):
        # 1. 初始化地图处理器
        self.mp = MapProcessor()
        
        # 2. 设置RRT*参数
        self.max_iter = 2000
        self.step_size = 0.5
        self.goal_threshold = 0.3
        self.search_radius = 1.0
        
        # 3. 创建发布器
        self.vis_path_pub = rospy.Publisher('/rrt_path', Path, queue_size=10)
        self.ctrl_path_pub = rospy.Publisher('/path', Float32MultiArray, queue_size=10)
        
        # 4. 订阅话题
        rospy.Subscriber('/map', OccupancyGrid, self.mp.map_callback)
        rospy.Subscriber('/move_base_simple/goal', PoseStamped, self.goal_callback)
        
        # 5. 等待地图加载
        rospy.wait_for_message('/map', OccupancyGrid)
```

### 步骤2: 路径规划执行
```python
def goal_callback(self, msg):
    self.goal = msg.pose.position
    self.start = Point(0, 0, 0)  # 起点固定为原点
    self.plan_path()            # 开始RRT*规划

def plan_path(self):
    # RRT*算法主循环
    for iteration in range(self.max_iter):
        # 1. 随机采样
        if random.random() < 0.1:
            rand_point = self.goal  # 10%概率采样目标点
        else:
            rand_point = self.sample_random_point()
        
        # 2. 找最近节点
        nearest_node = self.find_nearest_node(rand_point)
        
        # 3. 扩展新节点
        new_point = self.steer(nearest_node.point, rand_point)
        
        # 4. 碰撞检测
        if self.mp.is_collision(nearest_node.point, new_point):
            continue
            
        # 5. 选择最佳父节点
        near_nodes = self.find_near_nodes(new_point)
        best_parent = self.choose_parent(near_nodes, new_point, nearest_node)
        
        # 6. 添加节点并重连接
        new_node = self.Node(new_point, best_parent)
        self.nodes.append(new_node)
        self.rewire(new_node, near_nodes)
        
        # 7. 检查是否到达目标
        if self.reached_goal(new_point):
            goal_node = self.Node(self.goal, new_node)
            self.publish_path(goal_node)
            break
```

### 步骤3: 路径跟踪执行
```python
def path_callback(data):
    # 解析路径数据
    path_nodes = []
    for i in range(0, len(data.data), 2):
        x, y = data.data[i], data.data[i+1]
        path_nodes.append((x, y))
    
    # 创建控制器
    controller = RobotController(path_nodes)
    controller.move_robot()

def move_robot(self):
    while not rospy.is_shutdown() and self.current_target_idx < self.num_of_points:
        # 1. 更新目标点
        self.update_target_point()
        
        # 2. 计算控制命令
        linear_vel, angular_vel = self.calculate_control_commands()
        
        # 3. 速度平滑
        linear_vel, angular_vel = self.smooth_velocity(linear_vel, angular_vel)
        
        # 4. 发布控制指令
        move_cmd = Twist()
        move_cmd.linear.x = linear_vel
        move_cmd.angular.z = angular_vel
        self.cmd_vel_pub.publish(move_cmd)
```

## 核心数据流

### 话题通信架构
```
/map (OccupancyGrid)
    ↓
MapProcessor ← RRTStarPlanner
    ↓
/move_base_simple/goal (PoseStamped)
    ↓
RRT*算法执行
    ↓
/path (Float32MultiArray) + /rrt_path (Path)
    ↓
RobotController
    ↓
/cmd_vel (Twist)
    ↓
TurtleBot3机器人
    ↓
/odom (Odometry)
    ↓
RobotController (状态反馈)
```

### 坐标系转换
```python
# 世界坐标 → 地图坐标
def world_to_map(self, point):
    x = int((point.x - self.map_info.origin.position.x) / self.map_info.resolution)
    y = int((point.y - self.map_info.origin.position.y) / self.map_info.resolution)
    return (x, y)

# 四元数 → 欧拉角
def quat_2_euler(ox, oy, oz, ow):
    t3 = +2.0 * (ow * oz + ox * oy)
    t4 = +1.0 - 2.0 * (oy * oy + oz * oz)
    yaw_z = math.atan2(t3, t4)
    return yaw_z
```

## 性能优化机制

### 1. RRT*算法优化
- **目标偏向采样**: 10%概率直接采样目标点，提高收敛速度
- **重连接优化**: 持续优化路径，降低总代价
- **动态搜索半径**: 根据节点密度调整搜索范围

### 2. 碰撞检测优化
- **Bresenham算法**: O(n)时间复杂度的直线碰撞检测
- **早期终止**: 一旦检测到碰撞立即返回

### 3. 控制系统优化
- **前瞻控制**: 根据前瞻距离选择目标点
- **速度平滑**: 滑动平均滤波器减少速度突变
- **自适应减速**: 接近目标时自动减速

```python
# 最大迭代次数保护
for iteration in range(self.max_iter):
    # ... RRT*算法
    if iteration >= self.max_iter - 1:
        rospy.logwarn("路径规划超时，未找到解")
```

### 4. 控制器异常
```python

self.integral_error = max(-self.integral_limit, 
                         min(self.integral_limit, self.integral_error))
```
