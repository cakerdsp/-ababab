```python
def extract_obstacles(self):
    """从占据栅格地图中提取障碍物坐标"""
    self.obstacles = []
    for y in range(self.map_info.height):
        for x in range(self.map_info.width):
            # 占据概率大于50%的认为是障碍物
            if self.map_data[y][x] > 50:
                world_x = x * self.map_info.resolution + self.map_info.origin.position.x
                world_y = y * self.map_info.resolution + self.map_info.origin.position.y
                self.obstacles.append((world_x, world_y))
```

```python
for iteration in range(self.max_iter):
    # 1. 随机采样 (10%概率采样目标点)
    if random.random() < 0.1:
        rand_point = Point(self.goal.x, self.goal.y, 0)
    else:
        rand_point = self.sample_random_point()
    
    # 2. 找到最近的节点
    nearest_node = self.find_nearest_node(rand_point)
    
    # 3. 扩展节点
    new_point = self.steer(nearest_node.point, rand_point)
    
    # 4. 碰撞检测
    if self.mp.is_collision(nearest_node.point, new_point):
        continue
    
    # 5. RRT*优化步骤
    near_nodes = self.find_near_nodes(new_point)
    best_parent = self.choose_parent(near_nodes, new_point, nearest_node)
    new_node = self.Node(new_point, best_parent)
    self.nodes.append(new_node)
    self.rewire(new_node, near_nodes)
```

#### 路径发布
**第374-418行**:

```python
def publish_path(self, goal_node):
    """发布规划得到的路径"""
    # 可视化Path构建
    vis_path = Path()
    vis_path.header.frame_id = "map"
    
    # 控制用路径数据构建
    ctrl_path = Float32MultiArray()
    path_points = []
    
    # 收集路径点
    current = goal_node
    while current is not None:
        pose = PoseStamped()
        pose.header.frame_id = "map"
        pose.pose.position = current.point
        vis_path.poses.append(pose)
        path_points.append((current.point.x, current.point.y))
        current = current.parent
    
    # 反转路径顺序
    vis_path.poses.reverse()
    path_points.reverse()
    
    # 同时发布两种格式
    self.vis_path_pub.publish(vis_path)      # RViz可视化用
    self.ctrl_path_pub.publish(ctrl_path)    # 控制器用
```

**发布的消息**:
- `/rrt_path` (nav_msgs/Path): RViz显示红色路径线
- `/path` (std_msgs/Float32MultiArray): 控制器接收的路径点坐标
- `/rrt_tree` (visualization_msgs/MarkerArray): RViz显示绿色搜索树

**第100-156行**:
```python
class PIDController:
    def __init__(self, kp=0.0, ki=0.0, kd=0.0, integral_limit=1.0):
        self.kp = kp                    # 比例增益
        self.ki = ki                    # 积分增益
        self.kd = kd                    # 微分增益
        self.integral_limit = integral_limit  # 积分限幅
        
    def update(self, error, dt=None):
        """更新PID控制器"""
        # 比例项
        proportional = self.kp * error
        
        # 积分项（带限幅）
        self.integral_error += error * dt
        self.integral_error = max(-self.integral_limit, 
                                min(self.integral_limit, self.integral_error))
        integral = self.ki * self.integral_error
        
        # 微分项
        derivative = self.kd * (error - self.error_last) / dt
        
        return proportional + integral + derivative
```

#### 控制命令计算
**第234-278行**:
```python
def calculate_control_commands(self):
    """计算控制命令（线速度和角速度）"""
    target_point = self.get_target_point()
    
    # 计算位置误差
    dx = target_point[0] - self.current_pos[0]
    dy = target_point[1] - self.current_pos[1]
    distance_error = math.sqrt(dx*dx + dy*dy)
    
    # 计算目标角度
    target_angle = math.atan2(dy, dx) * 180 / math.pi
    
    # 计算角度误差
    angle_error = normalize_angle((target_angle - self.current_orient) * math.pi / 180)
    
    # PID控制
    linear_velocity = self.linear_pid.update(distance_error)
    angular_velocity = self.angular_pid.update(angle_error)
    
    return linear_velocity, angular_velocity
```

**第321-363行 - 主控制循环**:

```python
def move_robot(self):
    """主控制循环，控制机器人沿路径移动"""
    rate = rospy.Rate(20)  # 20Hz控制频率
    move_cmd = Twist()
    
    while not rospy.is_shutdown() and self.current_target_idx < self.num_of_points:
        # 计算控制命令
        linear_vel, angular_vel = self.calculate_control_commands()
        
        # 速度平滑
        linear_vel, angular_vel = self.smooth_velocity(linear_vel, angular_vel)
        
        # 发布控制命令
        move_cmd.linear.x = linear_vel
        move_cmd.angular.z = angular_vel
        self.cmd_vel_pub.publish(move_cmd)
        
        rate.sleep()
```

## ⚡ 系统交互流程

### 完整数据流:
1. **用户点击** RViz → `/move_base_simple/goal` 
2. **路径规划器接收目标** → RRT*算法计算
3. **发布路径** → `/path` + `/rrt_path`
4. **控制器接收路径** → PID计算开始
5. **发布速度指令** → `/cmd_vel` (20Hz)
6. **Gazebo执行运动** → 差速驱动插件
7. **反馈位置信息** → `/odom` (30Hz)
8. **控制器更新** → 循环至目标点

### 关键性能参数:
- **规划算法**: RRT* (最大2000次迭代)
- **控制频率**: 20Hz
- **里程计频率**: 30Hz  
- **Gazebo更新**: 30Hz
- **前瞻距离**: 0.3米
- **位置容忍度**: 0.15米

这个系统实现了从用户交互到机器人运动的完整闭环控制，所有组件通过ROS话题进行通信，实现了高效的路径规划和精确的路径跟踪。
