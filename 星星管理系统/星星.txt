void MOVE(struct PLANET*p1,struct PLANET*p2)//p1,p2两个行星进行分析，然后更新v与a
{
	double theta,r,f;
	r=sqrt(((p1->x)-(p2->x))*((p1->x)-(p2->x))+((p1->y)-(p2->y))*((p1->y)-(p2->y)));
	f=G*(p1->m)*(p2->m)/(r*r);
	theta=atan(((p1->y)-(p2->y))/((p1->x)-(p2->x)));
	p1->ax=fabs(cos(theta)*(f)/(p1->m));
	p1->ay=fabs(sin(theta)*(f)/(p1->m));
	if((int)((p1->x)-(p2->x))>=0)
		{
			p1->vx= p1->vx-(p1->ax)*dt;
			p1->a0x=(p1->a0x)-(p1->ax);
	}
	else
	{
		p1->vx= p1->vx+(p1->ax)*dt;
		p1->a0x=(p1->a0x)+(p1->ax);
	}
	
	if((int)((p1->y)-(p2->y))>=0)
		{
			p1->vy=p1->vy-(p1->ay)*dt;
			p1->a0y=(p1->a0y)-(p1->ay);
	}
	else
		{p1->vy=p1->vy+(p1->ay)*dt;
	    p1->a0y=(p1->a0y)+(p1->ay);
	}
}
void MOVE2(struct PLANET*p)
{
	p->x=p->x+(p->vx)*dt;
	p->y=p->y+(p->vy)*dt;
}

void xiana(struct PLANET*p,double t)//画F函数线
{
	//画线
	int i;
	double a0;
	char shu[50];
	for(i=0;i<num;i++)
	{
	a0=sqrt(((p+i)->a0x)*((p+i)->a0x)+((p+i)->a0y)*((p+i)->a0y));
	sprintf(shu,"%.7lf10^(-3)m/s^2",a0);
	setcolor(BLACK);
	outtextxy(1050,(int)((-((a0))*q1+730)),shu);
	putpixel((int)((t)*4+1230),(int)((-(a0))*q1+730),RGB((p+i)->color[0],(p+i)->color[1],(p+i)->color[2]));
	}
}
void xianv(struct PLANET*p,double t)//画v函数线
{
	int i;
	double v;
	char shu[50];
	for(i=0;i<num;i++)
	{
	v=sqrt(((p+i)->vx)*((p+i)->vx)+((p+i)->vy)*((p+i)->vy));
	sprintf(shu,"%.4lf×10^3m/s",v);
	setcolor(BLACK);
	outtextxy(1050,(int)((-v)*q2+730),shu);
	putpixel((int)((t)*4+1230),(int)((-v)*q2+730),RGB((p+i)->color[0],(p+i)->color[1],(p+i)->color[2]));
	}
}
