#include<stdio.h>
#include<graphics.h>
#include<math.h>
#define G 5
#define dt 0.01
#define High 1000  // 游戏画面尺寸
#define Width 1000
int NUM=2;
struct PLANET
{
	double x;
	double y;
	double m;
	double vx;
	double vy;
	double ax;
	double ay;
	double r;
};
void MOVE(struct PLANET*p1,struct PLANET*p2)//p1,p2两个行星进行分析，然后更新v与a
{
 
 double theta,r,f,theta2;
 r=sqrt(((p1->x)-(p2->x))*((p1->x)-(p2->x))+((p1->y)-(p2->y))*((p1->y)-(p2->y)));
 f=G*(p1->m)*(p2->m)/(r*r);
 theta=atan(((p1->y)-(p2->y))/((p1->x)-(p2->x)));
 p1->ax=fabs(cos(theta)*(f)/(p1->m));
 //p2->ax=fabs(cos(theta)*(f)/(p2->m));
 if((int)((p1->x)-(p2->x))>=0)
 {
 p1->vx= p1->vx-(p1->ax)*dt;
 //p2->vx= p2->vx+(p2->ax)*dt;
 }
 else
 {
 p1->vx= p1->vx+(p1->ax)*dt;
 //p2->vx= p2->vx-(p2->ax)*dt;
 }
 //p1->x=p1->x+(p1->vx)*dt;
 //p2->x=p2->x+(p2->vx)*dt;
 //p1->y=p1->y+(p1->vy)*dt;
 //p2->y=p2->y+(p2->vy)*dt;
 theta2=atan(((p1->y)-(p2->y))/((p1->x)-(p2->x)));
 p1->ay=fabs(sin(theta2)*(f)/(p1->m));
 //p2->ay=fabs(sin(theta2)*(f)/(p2->m));
 if((int)((p1->y)-(p2->y))>=0)
 {
 p1->vy=p1->vy-(p1->ay)*dt;
 //p2->vy=p2->vy+(p2->ay)*dt;
 }
 else
 {
 p1->vy=p1->vy+(p1->ay)*dt;
// p2->vy=p2->vy-(p2->ay)*dt;
 }
}
void MOVE2(struct PLANET*p)
{
 p->x=p->x+(p->vx)*dt;
 p->y=p->y+(p->vy)*dt;
}

void multiply(struct PLANET*p)
{
	void MOVE(struct PLANET*p1,struct PLANET*p2);//p1,p2两个行星进行分析，然后更新v与a数值；
	void MOVE2(struct PLANET*p);
	int i,j,k;
	for(i=0;i<NUM;i++)
	{
		for(j=0;j<NUM;j++)
		{
			if(j=i)
				continue;
			else MOVE((p+i),(p+j));
		}
	}
  for(k=0;k<NUM;k++)
  {
	 MOVE2(p+k);//把dv数据加到x与y坐标上
	 setfillcolor(GREEN);
	 fillcircle((p+k)->x,(p+k)->y,(p+k)->r);
  }
}
