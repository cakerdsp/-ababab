char Low2Up(char c)//如果输入是小写字母，就转化为大写字母，否则不用管
{
if(c>='a'&&c<='z')
c=c-32;
return c;
}

char GetChar()//输入一个字符（非空格、非回车、ASCII码大于零）
{
char c;
do
{
c=getchar();
}
while(c=='\n'||c==' '||c<0);
return c;
}
