二：场景元素设计
Scene_id：场景id，服务端全局唯一
Scene_name: 场景名字，同一个用户同一个分组内唯一
actions: 场景动作集合
	action_id: 动作id，服务端全局唯一
	did: 设备id
	operation: 设备属性
	value: 操作值

五：客户端
客户端在用户新增和更新场景时，将新增和更新请求发往服务端前需要做智能电源补齐和动作冲突检测和动作属性值校验
注意需要提供有获取分组设备的接口函数，当用户设置场景动作时，点击设备栏，可以展开分组内可选设备
1.	智能电源补齐
1.	当场景包含 mode/fan_speed/target_temp 等依赖开机的属性时 → 自动在动作列表开头插入 power: true
2.	当场景包含 power: false 时 → 自动移除所有其他属性设置（避免无效操作）
3.	参考实现如下，需要开机的属性列表保存在单独文件中
 // 设备类型 -> 需要开机的属性列表
const POWER_DEPENDENT_PROPS = {
  mi_ac: ['mode', 'fan_speed', 'target_temp'],
  mi_humidfstr: ['target_humidity'],
  // ...其他设备类型
};

function autoFixSceneActions(actions) {
  const newActions = [...actions];
  const deviceTypes = new Set(actions.map(a => a.device_type));
  
  // 检查是否需要补开机指令
  deviceTypes.forEach(type => {
    const needsPower = newActions.some(a => 
      a.device_type === type &&
      POWER_DEPENDENT_PROPS[type]?.includes(a.property)
    );
    const hasPower = newActions.some(a => 
      a.device_type === type && a.property === 'power'
    );
    
    if (needsPower && !hasPower) {
      newActions.unshift({  // 插入到动作列表开头
        device_id: actions.find(a => a.device_type === type).device_id,
        device_type: type,
        property: 'power',
        value: true
      });
    }
  });
  return newActions;
}
// 如果有关机指令，移除其他无效设置

const hasPowerOff = newActions.some(a => a.property === 'power' && a.value === false);

if (hasPowerOff) {

return newActions.filter(a =>
  a.property === 'power' ||
  !POWER_DEPENDENT_PROPS[a.device_type]?.includes(a.property)
);

}

return newActions;
 
2.	动作属性值合理性校验
像手动控制一样，属性和属性对应值是否是合理数据类型，精度正确，范围正确
3.	动作冲突检测
定义冲突规则表，独立文件实现，遍历场景中属性，查冲突表，看看场景中有没有冲突的动作
参考实现如下：注意具体的冲突看手动控制文档设备属性确定的标黄部分(属性和属性本身冲突，也就是不能在一个场景内重复设置同一个设备的同一个属性)
 
const CONFLICT_RULES = {

// 格式：设备类型 -> { 冲突属性：[依赖属性的值] }

mi_ac: {
  'fan_speed': ['dry'],
  // dry模式下不可调风速
  'target_temp': ['fan'] // fan模式下不可调温度
},
mi_humidifier: {
  'target_humidity': ['sleep'] // sleep模式下不可调湿度
}

};
function checkConflict(deviceType, currentState, action) {
  const rules = CONFLICT_RULES[deviceType] || {};
  for (const [prop, forbiddenValues] of Object.entries(rules)) {
    if (action.property === prop && 
        forbiddenValues.includes(currentState.mode)) {
      return `[错误] ${prop}在${currentState.mode}模式下不可设置`;
    }
  }
  return true; // 无冲突
}
 
4.	其他
客户端场景信息更新策略
新建场景时，可以把新场景信息保存在内存，无需向服务端拉取新的场景列表和新场景详细信息
删除场景时，也是在本地也要删除场景信息，无需向服务端拉取新的场景列表和新场景详细信息
……
什么操作前都同步本地操作，这样就可以保持和服务端一致性，不需要周期性从服务端拉取场景信息（当然最好可以实现周期性更新，简化实现可以不管）


执行策略时：接受响应信息，最好可以反馈给用户场景执行状态
