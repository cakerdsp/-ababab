Use School
go
create procedure HHH
	@cid char(10),
	@cname varchar(30),
	@hour int,
	@ans varchar(100) out
as
	begin tran
		if not exists(select cid from COURSES where cid = @cid)
			begin
				select @ans = '课程信息不存在'
				goto error
			end
		update COURSES SET cname = @cname，hour = @hour where cid = @cid
		if @@ERROR > 0
			begin 
				select @ans = '修改课时失败'
				goto error
			end
		select @ans = '课时修改成功'
		print @ans
		commit tran
error:
	print @ans
	rollback tran
go



Use School
declare	@cid char(10)
declare	@cname varchar(30)
declare	@hour int
declare	@ans varchar(100) 
exec HHH '10001','database',0,@ans out
