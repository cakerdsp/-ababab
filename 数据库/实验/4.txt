use School;
with CHOICES_ as (
	select * 
	from CHOICES
	where CHOICES.cid = (
		select cid
		from COURSES	
		where cname = 'C++'
	)
)

select sid, sname
from STUDENTS S
where S.sid in(
	select sid 
	from CHOICES_
	where CHOICES_.score > (
		select score
		from CHOICES_
		where sid = (
			select sid
			from STUDENTS
			where STUDENTS.sname = 'ZNKOO'
		)
	)
)
